
const { EmbedBuilder, PermissionsBitField } = require('discord.js');
const fs = require('node:fs');
const path = require('node:path');

const statsConfigPath = path.join(__dirname, 'stats-config.json');

function lireStatsConfig() {
  try {
    return JSON.parse(fs.readFileSync(statsConfigPath, 'utf8'));
  } catch {
    return {};
  }
}

function sauverStatsConfig(config) {
  fs.writeFileSync(
    statsConfigPath,
    JSON.stringify(config, null, 2)
  );
}

// Crée l'embed des statistiques
async function creerEmbedStats(guild) {
  await guild.members.fetch();

  const membres = guild.memberCount;
  const enVocal = guild.members.cache.filter(
    m => m.voice.channelId
  ).size;

  const enLigne = guild.members.cache.filter(
    m => m.presence &&
      ['online', 'idle', 'dnd'].includes(m.presence.status)
  ).size;

  const actifs = guild.members.cache.filter(
    m => m.presence?.activities?.some(
      a => a.type === 0
    )
  ).size;

  const enStream = guild.members.cache.filter(
    m => m.voice.streaming
  ).size;

  const mute = guild.members.cache.filter(
    m => m.voice.channelId &&
      (m.voice.selfMute || m.voice.serverMute)
  ).size;

  const embed = new EmbedBuilder()
    .setColor(0x2b2d31)
    .setTitle(`Statistiques de ${guild.name} 🏯\n#NEW`)
    .setThumbnail(guild.iconURL({ size: 256 }))
    .setDescription(
      `**Membres**\n${membres}\n` +
      `**En ligne**\n${enLigne}\n` +
      `**En vocal**\n${enVocal}\n` +
      `**Actif**\n${actifs}\n` +
      `**En stream**\n${enStream}\n` +
      `**Mute**\n${mute}`
    )
    .setFooter({
      text: `${guild.name.toUpperCase()} • Statistiques du serveur`
    })
    .setTimestamp();

  return embed;
}

// Commande +stats #salon
client.on('messageCreate', async message => {
  if (!message.guild || message.author.bot) return;
  if (!message.content.startsWith('+stats')) return;

  if (!message.member.permissions.has(
    PermissionsBitField.Flags.ManageGuild
  )) {
    return message.reply('Tu dois avoir la permission Gérer le serveur.');
  }

  const salon = message.mentions.channels.first();

  if (!salon || !salon.isTextBased() || !salon.guild) {
    return message.reply('Utilise : +stats #salon');
  }

  try {
    const embed = await creerEmbedStats(message.guild);
    const statsMessage = await salon.send({ embeds: [embed] });

    const config = lireStatsConfig();
    config[message.guild.id] = {
      salonId: salon.id,
      messageId: statsMessage.id
    };
    sauverStatsConfig(config);

    return message.reply(
      'Statistiques configurées. Le bot publiera les statistiques chaque jour dans ce salon.'
    );
  } catch (error) {
    console.error(error);
    return message.reply(
      'Impossible de créer les statistiques. Vérifie les permissions du bot.'
    );
  }
});

// Publication quotidienne
let statsTimerLance = false;

client.once('ready', () => {
  if (statsTimerLance) return;
  statsTimerLance = true;

  setInterval(async () => {
    const config = lireStatsConfig();

    for (const [guildId, data] of Object.entries(config)) {
      try {
        const guild = await client.guilds.fetch(guildId);
        const salon = await guild.channels.fetch(data.salonId);

        if (!salon || !salon.isTextBased()) continue;

        const embed = await creerEmbedStats(guild);
        await salon.send({ embeds: [embed] });
      } catch (error) {
        console.error(`Erreur stats pour ${guildId}:`, error);
      }
    }
  }, 24 * 60 * 60 * 1000);
});
